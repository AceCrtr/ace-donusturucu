async function startConversion() { const btn = document.getElementById('actionBtn'); const fileInput = document.getElementById('fileInput'); const file = fileInput.files[0]; if(!file) { alert("Önce bir dosya seçmelisin!"); return; } btn.style.opacity = "0.5"; btn.innerHTML = "DÖNÜŞTÜRÜLÜYOR..."; btn.disabled = true; const formData = new FormData(); formData.append('file', file); try { // Senin Render sunucuna bağlanıyoruz const response = await fetch('https://ace-donusturucu.onrender.com/convert', { method: 'POST', body: formData }); if (response.ok) { const blob = await response.blob(); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = "ace-donusturucu.pdf"; document.body.appendChild(a); a.click(); a.remove(); } else { alert("Sunucu bir hata verdi. Dosya formatını kontrol et."); } } catch (error) { console.error(error); alert("Sunucuya bağlanılamadı. Render uyanıyor olabilir, lütfen 30 saniye sonra tekrar dene."); } finally { btn.style.opacity = "1"; btn.innerHTML = "ŞİMDİ DÖNÜŞTÜR"; btn.disabled = false; } }
